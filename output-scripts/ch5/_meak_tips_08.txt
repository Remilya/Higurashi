	ClearMessage();

	DrawScene("background/nort", 1000 );
	PlayBGM( 0, "z2_hig1", 56, 0 );

	if (GetGlobalFlag(GADVMode)) { OutputLineAll("", NULL, Line_ContinueAfterTyping); }
	OutputLine(NULL, "　悟史くんが東京へ行った、という怪情報の裏付けは全く取れない。",
		   NULL, "Satoshi-kun'un Tokyo'ya gitmiş olabileceğine dair o şüpheli bilgi bir türlü doğrulanamadı.", Line_Normal);
	ClearMessage();
	PlaySE( 0, "s_page", 56, 64 );

	OutputLine(NULL, "　そもそも情報の発端はこうだ。",
		   NULL, "Söylentinin çıkış noktası şöyleydi:", Line_Normal);
	ClearMessage();
	DrawScene("black", 1000 );
	DrawSceneWithMask("background/eki", "maskup", 1, 0, 1300 );

	if (GetGlobalFlag(GADVMode)) { OutputLineAll("", NULL, Line_ContinueAfterTyping); }
	OutputLine(NULL, "　名古屋駅の遺失物窓口に若者が訪れて、自分の財布が届けられてないかと騒いだのだ。",
		   NULL, "Genç bir adam Nagoya İstasyonu'ndaki kayıp eşya gişesine gelip cüzdanının bulunup bulunmadığını sormuş.", GetGlobalFlag(GLinemodeSp));

	OutputLine(NULL, "　それで駅の職員が、届け出がないか調べてきますからその間にこちらにお名前を書いてください、と用紙を渡した。",
		   NULL, "İstasyon görevlisi de kayıtlara bakacağını söyleyerek ondan forma adını yazmasını istemiş.", Line_Normal);
	ClearMessage();

	if (GetGlobalFlag(GADVMode)) { OutputLineAll("", NULL, Line_ContinueAfterTyping); }
	OutputLine(NULL, "　そうしたらその若者は、北条と書きかけてからその用紙をくしゃくしゃにして捨て、もう１枚の用紙に全然違う名前を書いた、というのだ。",
		   NULL, "Bunun üzerine genç adam forma Houjou yazmaya başlamış, ama yarıda durup kağıdı buruşturup atmış ve başka bir forma bambaşka bir isim yazmış.", GetGlobalFlag(GLinemodeSp));
